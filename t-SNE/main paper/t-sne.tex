\section{Introduction} \label{sec_intro}
The t-distributed stochastic neighbour embedding(t-SNE) is a well-known method of visualizing high-dimensional data with low-dimensional vectors (so they can be easily visualized). Let $d,s \geq 2$ with $d>s$, and $n \geq 1$. The input of t-SNE consist of vectors $X_1, \cdots, X_n \in \mathbb{R}^d$. The goal is the find vectors $Y_1, \cdots Y_n \in \mathbb{R}^s$ that serve as a good representation of the original vectors. The process of t-SNE starts with defining the following probability masses:
\begin{align} \label{original_prob_mass}
    p_{j|i} = \frac{\exp(-\norm{X_i-X_j}^2/2\sigma_i^2)}{\sum_{k \neq i} \exp(-\norm{X_i-X_k}^2/2\sigma_i^2)},
\end{align}
where $\sigma_i$ are given constants. Notice that we assigned a Gaussian mass to each pair of input vector. Or we can say in the original t-SNE, the algorithm has a \textit{Gaussian kernel}. In this paper, we want to normalize t-SNE with generalized result. Let $f(x,y): \mathbb{R}^d  \times \mathbb{R}^d \to \mathbb{R}^+$ be such that $f(x,y)=f(y,x)$. We replace the probability mass in (\ref{original_prob_mass}) with the following
\begin{align}
    p_{j|i} = \frac{f(X_i,X_j)}{\sum_{k \neq i} f(X_i,X_k)}.
\end{align}
We call $f(x,y)$ \textit{the generalized kernel} of the t-SNE method. Now we set up the problem like in the original setting. Let $p_{ii} = 0$ and for $1 \leq i \neq j \leq n$,
\begin{align}
    p_{ij} = \frac{p_{j|i} + p_{i|j}}{2n} = \frac{1}{2n}\left(\frac{f(X_i,X_j)}{\sum_{k \neq i} f(X_i,X_k)}+\frac{f(X_i,X_j)}{\sum_{k \neq j} f(X_j,X_k)}\right).
\end{align}
Both $p_{j|i}$ and $p_{ij}$ are well-defined probability masses as obviously,
\begin{align}
    \sum_{\substack{j = 1 \\ j \neq i}}^n p_{j|i} = 1, \quad \sum_{1 \leq i \neq j \leq n} p_{ij} = 1.
\end{align}
As for the output vectors $Y_1, \cdots Y_n \in \mathbb{R}^s$, set for $1 \leq i \neq j \leq n$,
\begin{align}
    q_{ij} = \frac{(1+\norm{Y_i-Y_j}^2)^{-1}}{\sum_{k \neq l}(1+\norm{Y_k-Y_l}^2)}.
\end{align}
The definition of $q_{ij}$ is the same as the original. Denote $X = (X_1,\cdots,X_n)$ and $Y = (Y_1,\cdots,Y_n)$. The goal of t-SNE is to find $Y$ that minimizes the following quantity:
\begin{align}
    L_n(X,Y) = \sum_{1 \leq i \neq j \leq n} p_{ij} \log \left(\frac{p_{ij}}{q_{ij}}\right).
\end{align}
More precisely, the output of t-SNE $Y^*$ is defined as follows:
\begin{align}
    Y^* = \argmin_{Y \in (\mathbb{R}^s)^n}L_n(X,Y).
\end{align}
In the expression of $L_n(X,Y)$, all $p_{ij}$ are given and $q_{ij}$ is a function of $Y_i$ and $Y_j$. Therefore, $Y^*$ can be found simply by using gradient descent. This is no different from the original version of t-SNE. However, it is the mathematical properties and implications of t-SNE that we are interested in. Towards this end, let $\mathcal{M}(\mathbb{R}^d)$ be the space of Borel probability measures on $\mathbb{R}^d$ and $\mu_X \in \mathcal{M}(\mathbb{R}^d)$ be a probability measure. Suppose all input $X = (X_1,\cdots,X_n)$ are i.i.d., independent random variables with probability measure $\mu_X$. Meanwhile, let $Y^* = (Y_1,\cdots,Y_n)$ be the output of the t-SNE algorithm. Consider the following empirical measure:
\begin{align}
    \mu_n := \frac{1}{n} \sum_{i=1}^n \delta(X_i,Y_i) \in \mathcal{M}(\mathbb{R}^d \times \mathbb{R}^s).
\end{align}
We are interested in its limiting measure as $n \to \infty$. To finish the setup, there are a few more remaining quantity we would like to introduce. Let $\mu \in \mathcal{M}(\mathbb{R}^d \times \mathbb{R}^s)$. For a given kernel $f(x,y)$, define $F_{\mu,f} : \mathbb{R}^d \to \mathbb{R}$ as
\begin{align}
    F_{\mu,f}(x) = \int \frac{f(x,x')}{\int f(x,x')d\mu(x')} \log \left(\frac{f(x,x')}{\int f(x,x')d\mu(x')}\right) d\mu(x').
\end{align}
Note that the last $s$ variables in the integrands stay constant.

Let $q: \mathbb{R}^d \times \mathbb{R}^d \to \mathbb{R}$ be defined by
\begin{align}
    p(x,x') = \frac{1}{2}\left( \frac{f(x,x')}{\int f(x,x')d\mu(x')} + \frac{f(x,x')}{\int f(x,x')d\mu(x)}\right).
\end{align}
Then, let $q: \mathbb{R}^s \times \mathbb{R}^s \to \mathbb{R}$ be given by
\begin{align}
    q(y,y') = \frac{(1+\norm{y-y'}^2)^{-1}}{\iint (1+\norm{y-y'}^2)^{-1}d\mu(y) d\mu(y')}.
\end{align}
Based on the construction of $p$ and $q$, define
\begin{align}
    I(\mu) = \iint p(x,x') \log\left(\frac{p(x,x')}{q(y,y')}\right) d\mu(x,y) d\mu(x',y').
\end{align}
Recall $\mu_X \in \mathcal{M}(\mathbb{R}^d)$ is an arbitrary given probability measure. We set $\mathcal{P}_X$ to be the set of probability measures on $\mathbb{R}^d \times \mathbb{R}^s$ whose marginal coincides with $\mu_X$. Formally speaking,
\begin{align}
    \mathcal{P}_X := \{\mu \in \mathcal{M}(\mathbb{R}^d \times \mathbb{R}^s): \text{the $\mathbb{R}^d$ marginal of $\mu$ is $\mu_X$}\}.
\end{align}
Lastly, define
\begin{align}
    \tilde{\mathcal{P}}_X := \left\{ \mu \in \mathcal{P}_X: \int (p(x,x')-q(y,y'))\frac{y-y'}{1+\norm{y-y}^2} d\mu(x',y') = 0,\mu(x,y)\text{-a.e.}\right\}.
\end{align}
The goal of this project is to find the relationship between the following two quantities:
\begin{align}
    \lim_{n \to \infty} \inf_Y L(X,Y), \text{ and} \inf_{\mu \in \tilde{\mathcal{P}}_X} I(\mu).
\end{align}

\section{First Estimation and Condition Tweaks}
We follow the work and deduction in \cite{auffinger-tsne} to start our estimation. Until the completion of this project, the conditions we impose on the kernel are subject to change, as we are basically doing reverse engineering on the conditions to make sure it carries out the results we desire. There will be a few parts that we can borrow directly from \cite{auffinger-tsne} while others we have to tweak through trails and errors.


First off are the assumptions on the samples, a.k.a., random variables $X_i$'s. Same as the original paper, we will assume each $X_i$ to be i.i.d. with sub-Gaussian tails. We review the meaning of sub-Gaussian tails by giving out the following formal definition:
\begin{definition}
    A probability $\nu$ on $\mathbb{R}$ is said to have sub-Gaussian tails if there exists a positive constant $C$ such that
    \begin{align}
        \nu((t,\infty)) \leq 2 \exp(-Ct^2).
    \end{align}
\end{definition}
What follows are a few lemmas from \cite{auffinger-tsne} that discuss the properties of our samples $X_i$. Since they do not depend on the kernel $f(x,y)$.
\begin{lemma} \label{sample_lemma_1}
    Let $\nu$ be a sub-Gaussian with second moment equal to 1 and $X = (X_i)_{i \leq d} \in \mathbb{R}^d$ be i.i.d. random variables with probability measure $\nu$. Then there exists constant $C,c_1$ such that
    \begin{align}
        \p(\norm{X} \geq \sqrt{d}+t) \leq Ce^{-c_1t^2}.
    \end{align}
\end{lemma}
\begin{lemma}
    Let $nu$ and $c_1$ be the same as introduced in lemma \ref{sample_lemma_1}. Let $X_1, \cdots, X_n \in \mathbb{R}^d$ be i.i.d random variables with probability measure $\nu$, Then there exists a constant $C$ such that
    \begin{align}
        \p \left(\max_{1\leq i \leq n}\norm{X_i} \geq \sqrt{\frac{\log n}{c_1}} + \sqrt{d} +t\right) \leq Ce^{-c_1t^2}.
    \end{align}
\end{lemma}
\begin{lemma} \label{kernel_condition_lemma}
    Let $\delta>0$ be a constant. $K_n = [\delta, \sqrt{n}]$ and $h : \mathbb{R}^d \times \mathbb{R}^d \times \mathbb{R} \to \mathbb{R}$. Suppose the following hold:
    
    1. $\mu$ is a sub-Gaussian probability measure;

    2. $h(x,x',\sigma)$ is $C^{1}$ for all $x,x'$ and all $\sigma \in K_n$;

    3. $h(x,x',\sigma)$ and $\nabla_x h(x,x',\sigma)$ are bounded for all $x,x'$ and all $\sigma \in K_n$;

    4. $h(x,x',\sigma) \to 0$ as $\norm{x-x}/\sigma \to \infty$.
    Let $X_1, \cdots, X_n$ be i.i.d. random variables with probability measure $\mu$ and $\mu_n = \frac{1}{n} \sum_i \delta_{X_i}$. Then for any $\epsilon >0$ with probability no smaller than $1-e^{-c \epsilon^2 n}$ we have
    \begin{align}
        \sup_{\sigma \in K_n , x} \left|\int h(x,x',\sigma) d \mu_n(x') - \int h(x,x',\sigma) d \mu(x') \right| \leq \epsilon + O \left(\sqrt{\frac{log n}{n}}\right).
    \end{align}
\end{lemma}
The above lemma has a relatively loose restriction on the kernel. Indeed, in the original t-SNE the kernel is smooth with exponential decay. Moreover, the parameter $\sigma$ in the lemma is used to control the entropy of probability distribution $p_{j|i}$ for each $i$. We start to give our first set of conditions on the kernel $f(x,y)$.
\begin{condition}
    $f(x,y)$ is a function of $\norm{x-y}$. Moreover, $f(x,y) \equiv f(\norm{x-y})$ satisfies all conditions in the assumptions of lemma \ref{kernel_condition_lemma}.
\end{condition}
