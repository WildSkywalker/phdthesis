\section{Setup}
This document is dedicated to the reconstruction of \cite{auffinger-tsne} with low-dimension kernel $g$ replaced by a more general kernel. This would serve as a great way to understand the material and help with the completion of the original project. For the sake of simplicity, we will use the exact same notation as in \cite{auffinger-tsne} and skip all the definitions. Recall in \cite{auffinger-tsne}, $g$ is defined as 
\begin{align}
    g(y,y') = \frac{1}{1+\norm{y-y'}^2}
\end{align}
With all these being said, we will first start by giving our general low dimension kernel $g$ some general assumptions, and modify it as we move on. We first require that $g(y,y') = k (\norm{y-y'})$ such that:

1. $k: \mathbb{R}^+ \to \mathbb{R}^+$ is decreasing.

2. $k$ satisfies
\begin{align}
    \iint_{(0,\infty)^2} k(x,x') dxdx' > \infty
\end{align}
We will kick off this warmup by listing all the important lemmas and results in order, and prove our own versions when the need arises. The final goal is to prove a same set of results as in \cite{auffinger-tsne} with the $y$ kernel replaced.
Recall that
\begin{align}
    F_{\rho, \mu}(x,\sigma) = &- \int \frac{\norm{x-x'}^2}{2\sigma^2} \frac{\exp(-\norm{x-x'}^2/2\sigma^2)}{\int \exp(-\norm{x-x'}^2/2\sigma^2)d\mu(x')}d\mu(x') \nonumber \\
    &- \log \left(\int \exp(-\norm{x-x'}^2/2\sigma^2)\right) +\log \rho.
\end{align}
\begin{proposition}
    If $\mu$ has a $C^1$ density $f$ such that both $f$ and $\norm{\nabla f}$ are bounded, then the equation $F_{\rho, \mu}(x,\sigma)=0$ has a unique solution $\sigma^*_{\rho,\mu}(x)$ for all $\rho \in (0,1)$ and $x \in \mathbb{R}^d$. Moreover, $x \mapsto \sigma^*_{\rho,\mu}(x)$ is smooth.
\end{proposition}
\begin{proof}
    This proposition is about the kernel of $x$. The proof stays unchanged from \cite{auffinger-tsne}.
\end{proof}
The next couple of lemmas are general results in section 3 of \cite{auffinger-tsne}. We will omit the proof because they are not directly discussing the quantities in the t-sne algorithm.  Recall that $X$ or the corresponding probability measure $\mu$ is called sub-Gaussian if $\p(X>t) \leq 2 \exp(-Ct^2)$ for some positive constant $C$.
\begin{lemma} \label{Gaussian_lemma}
    Let $\nu$ be sub-Gaussian with second moment equal to 1 and suppose $X \in \mathbb{R}^d$ has i.i.d. entries drawn from $\nu$. Then there exists constant $C, c_1$ such that
    \begin{align}
        \p(\norm{X} \geq \sqrt{d} + t) \leq Ce^{-c_1t^2}.
    \end{align}
\end{lemma}
\begin{lemma} \label{max_iid_lemma}
Let $\mu$ be sub-Gaussian and $c_1$ be the constant from lemma \ref{Gaussian_lemma}. Let $X_1, \dots, X_n \in \mathbb{R}^d$ have i.i.d. entries drawn from $\mu$. Then there exists a constant $C$ such that
\begin{align}
    \p \left (\max_{i=1,\dots,n}\norm{X_i} \geq \sqrt{\frac{\log n}{c_1}}+\sqrt{d}+t \right ) \leq Ce^{-c_1t^2}.
\end{align}
\end{lemma}
\begin{lemma} \label{approx_lemma}
    Let $\delta>0$ be constant, $K_n = [\delta,\sqrt{n}]$ and $h: \mathbb{R}^d \times \mathbb{R}^d \times \mathbb{R} \to \mathbb{R}$. Suppose the following hold: \\
    1. $\mu$ is sub-Gaussian; \\
    2. $h(x,x',\sigma)$ and $\nabla_x h(x,x',\sigma)$ are bounded for al $x,x'$ and all $\delta \in K_n$;\\
    3. $h(x,x',\sigma) \to 0$ as $\norm{x-x'}/\sigma \to \infty$.\\
    Let $X_1,\dots,X_n$ be i.i.d. random variables drawn from $\mu$. Let $\mu_n = (1/n)\sum_i \delta_{X_i}$. Then for any $\epsilon > 0$ with probability at least $1 - e^{-c\epsilon^2n}$ we have
    \begin{align}
        \sup_{\sigma \in K_n,x} \left|\int h(x,x',\sigma)d\mu_n(x) - \int h(x,x',\sigma)d\mu(x')\right| \leq \epsilon + O\left(\sqrt{\frac{\log n}{n}}\right).
    \end{align}
\end{lemma}
Define
\begin{align}
    f_{\sigma}(x,x') = \exp(-\norm{x-x}^2/2\sigma^2), \quad g_{\sigma} = \frac{\norm{x-x}^2}{2\sigma^2} \exp(-\norm{x-x'}^2/2\sigma^2).
\end{align}
The next two lemmas use similar techniques to prove.
\begin{lemma}
    Let $c_1$ be the decay constant in lemma \ref{Gaussian_lemma} and let $\delta >0$. Then almost surely in $X_1,\dots,X_n$,
    \begin{align}
        \sup_{\substack{\sigma \in [\delta, \sqrt{n}] \\ i = 1,\dots,n}}\left|\frac{\int g_{\sigma}(X_i,x')d\mu_n(x')}{\int f_{\sigma}(X_i,x')d\mu_n (x')} - \frac{\int g_{\sigma}(X_i,x')d\mu(x')}{\int f_{\sigma}(X_i,x')d\mu (x')}\right| = O(n^{8/c_1 \delta^2 -1/2}\sqrt{\log n}).
    \end{align}
\end{lemma}
\begin{lemma}\label{pair_lemma_2}
    Let $c_1$ be the decay constant in lemma $\ref{Gaussian_lemma}$ and let $\delta>0$ be such that $c_1\delta^2>8$. Then almost surely in $X_1,\dots,X_n$,
    \begin{align}
        \sup_{\substack{\sigma \in [\delta, \sqrt{n}] \\ i = 1,\dots,n}} \left|\log \left(\int f_{\sigma}(X_i,x')d\mu_n (x')\right) - \log \left(\int f_{\sigma}(X_i,x')d\mu (x')\right)\right| = O(n^{4/c_1\delta^2 -1/2}\sqrt{\log n})
    \end{align}
\end{lemma}
For this pair of lemmas, \cite{auffinger-tsne} provides the proof to the first lemma. We will provide the second for the sake of exercise.
\begin{proof}[Proof of lemma \ref{pair_lemma_2}]
    First, notice that for arbitrary $x>0$, the following elementary inequality holds:
    \begin{align}
        \log(x) \leq x-1.
    \end{align}
    Therefore, the lemma can be proved if we are able to show that
    \begin{align}
        \left|\frac{\int f_{\sigma}(X_i,x')d\mu_n (x')}{\int f_{\sigma}(X_i,x')d\mu (x')} - 1 \right| =& \left|\frac{\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')}{\int f_{\sigma}(X_i,x')d\mu (x')} \right|\nonumber \\
        =& O(n^{4/c_1\delta^2 -1/2}\sqrt{\log n}).
    \end{align}
    We can achieve that by bounding the numerator and the denominator of the previous formula. On one hand, consider set
    \begin{align}
        E_1' = \left \{ \max_{i=1,\dots,n}\norm{X_i} \geq \sqrt{\alpha+\frac{1}{c_1}} \sqrt{\log n}+\sqrt{d}\right \}
    \end{align}
    Setting $t = \sqrt{\alpha \log n}$ in lemma \ref{max_iid_lemma}, we have $\p(E_1') \leq Cn^{-c_1\alpha}$. To absorb the $\sqrt{d}$ term, we can find $\epsilon_n \to 0$ such that
    \begin{align}
        E_1 = \left \{ \max_{i=1,\dots,n}\norm{X_i} \geq \sqrt{\alpha+\frac{1}{c_1}+\epsilon_n} \sqrt{\log n}\right \}
    \end{align}
    with $\p(E_1) \leq Cn^{-c_1\alpha}$.
    Because the maximum of norm of each $X_i$ is restricted, on $E_1^c$ we have for all $i,j$, that
    \begin{align}
        \norm{X_i-X_j}^2 \leq 4 \left(\alpha + \frac{1}{c_1} +\epsilon_n\right) \log n =:4\gamma \log n.
    \end{align}
    Then for $\sigma \in [\delta, \sqrt{n}]$, on $E_1^c$ for all $i$ we have
    \begin{align} \label{log_esti_1}
        \int f_{\sigma}(X_i,x')d\mu(x') \geq (1-Cn^{-c_1\alpha})\exp \left(-\frac{2\gamma \log n}{\delta^2}\right) \geq \frac{1}{2} \exp \left(-\frac{2\gamma \log n}{\delta^2}\right) =\frac{1}{2}n^{-2\gamma/\delta^2}.
    \end{align} 
    This helps us bound the denominator. On the other hand, by lemma \ref{approx_lemma}, we are able to find set $E_2$ with $\p(E_2) \leq e^{-c\epsilon^2 n}$, such that, on $E^c_2$,
    \begin{align} \label{log_esti_2}
        \sup_{\substack{\sigma \in [\delta,\sqrt{n}] \\ i = 1,\dots,n}} \left|\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')\right| \leq \epsilon + O\left(\frac{\sqrt{\log n}}{\sqrt{n}}\right).
    \end{align}
    Therefore, we can combine (\ref{log_esti_1}) and (\ref{log_esti_2}) and conclude that on $E_1^c \cap E_2^c$,
    \begin{align}
        \left|\frac{\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')}{\int f_{\sigma}(X_i,x')d\mu (x')} \right| \leq C\epsilon n^{2\gamma/\delta^2}+O(\sqrt{\log n} \cdot n^{-1/2 + 2\gamma/\delta^2}).
    \end{align}
    In order for the RHS to converge to zero, we need
    \begin{align}
        \epsilon n^{2\gamma/\delta^2} \to 0, \quad 1/2 - 2\gamma/\delta^2 > 0.
    \end{align}
    Moreover, we have $\p(E_1 \cup E_2) \leq Cn^{-c_1\alpha} + e^{-c \epsilon^2 n}$. Set $\epsilon = n^{-\eta}$ for some $\eta >0$. In order to get almost sure convergence using Borel-Cantelli, we also need
    \begin{align}
        c_1 \alpha>1, \quad 1-2\eta>0.
    \end{align}
    Hence for the entire lemma to work we need the following four inequalities to hold:
    \begin{align}
        \frac{2\gamma}{\delta^2}-\eta<0, \quad \frac{2\gamma}{\delta^2}-1/2<0, \quad c_1 \alpha>1, \quad 1-2\eta>0.
    \end{align}
    The last inequality implies that $\eta<1/2$ so the first inequality implies the second. Now, recall that
    \begin{align}
        \gamma := \alpha + \frac{1}{c_1} +\epsilon_n.
    \end{align}
    The collection of inequalities is equivalent to 
    \begin{align} \label{inequ_collection}
        2\left(\alpha + \frac{1}{c_1} +\epsilon_n\right)<\eta \delta^2, \quad c_1 \alpha>1, 0<\eta<1/2.
    \end{align}
    Here, we are free to choose $\alpha$ and $\eta$, but $c_1$ and $\delta$ are fixed. Now let
    \begin{align}
        \alpha = \frac{1}{c_1}+\epsilon, \quad \eta = \frac{1}{2} - \epsilon.
    \end{align}
    Then the second and third inequalities in (\ref{inequ_collection}) are satisfied. The first is equivalent to
    \begin{align}
        \frac{4}{c_1 \delta^2}+\epsilon<\frac{1}{2}-\epsilon
    \end{align}
    as $\epsilon_n \to 0$ as $n \to \infty$. In addition, since $\epsilon$ is arbitrary, we have again, equivalently,
    \begin{align}
        \frac{4}{c_1 \delta^2} - \frac{1}{2}<0.
    \end{align}
    When this is satisfied, we have:
    \begin{align}
        \sup_{\substack{\sigma \in [\delta,\sqrt{n}] \\ i = 1,\dots,n}} \left|\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')\right| =O(n^{4/c_1\delta^2 -1/2}\sqrt{\log n}).
    \end{align}
    The proof is thus finished.
\end{proof}
Then combining the previous result we have an important convergence:
\begin{proposition}
    The $\mu$ be sub-Gaussian and $c_1$ be the constant from lemma \ref{Gaussian_lemma}. Let $X_1,\dots, X_n$ be i.i.d. random variables with probability measure $\mu$ and $\mu_n = \frac{1}{n} \sum \delta_{X_i}$ be the empirical measure. Let $\delta>0$ be a constant such that $c_1 \delta^2>16$. Then, there exists $\epsilon>0$, such that almost surely in $X_1,\dots,X_n$:
    \begin{align}
        \sup_{\substack{\sigma \in [\delta,\sqrt{n}]\\i=1,\dots,n}}|F_{\mu_n}(X_i,\sigma)-F_{\mu}(X_i,\sigma)| = O(n^{-\epsilon}).
    \end{align} 
\end{proposition}
Let us summarize the important results upon this point, which will be used later:

There exists $\gamma = 2/c_1 +\epsilon$ such that the following holds almost surely

(1)$\max_{i=1,\dots,n}\norm{X_i} \leq \sqrt{\gamma \log n}$;

(2)$\max_{i,j}\norm{X_i-X_j}^2 \leq 4 \gamma \log n$;

(3)For $\sigma\in [\delta,\sqrt{n}]$,
\begin{align}
    \int \exp(-\norm{X_i-x'}^2/2\sigma^2)d\mu_n(x),\int \exp(-\norm{X_i-x'}^2/2\sigma^2)d\mu(x') \geq Cn^{-2\gamma/\sigma^2}.
\end{align}
Next couple of lemmas in \cite{auffinger-tsne} are about lower about on $\sigma^*_{\rho,\mu}$. Recall that $\sigma^*_{\rho,\mu}$ is the unique solution of the equation $F_{\rho,\mu}(x,\sigma^*_{\rho,\mu})=0$. For simplicity of notation, we write $\sigma^*(x) = \sigma^*_{\rho,\mu}(x)$ and $\sigma^*_n(x) = \sigma^*_{\rho,\mu_n}(x)$.
\begin{lemma}
    If $\mu$ satisfies the condition in lemma 1 of \cite{auffinger-tsne}, then $\sigma^*(x)$ and $\sigma^*_n(x)$ are uniformly bounded from below in both $i$ and $n$ almost surely.
\end{lemma}
\begin{proposition}
    Suppose $\mu$ has compact support and $C^1$ density. Then almost surely, as $n \to \infty$,
    \begin{align}
        \sup_{i=1,\dots,n}|\sigma_n^*(X_i)-\sigma^*(X_i)| \to 0.
    \end{align}
\end{proposition}
\begin{proof}
    By \cite{auffinger-tsne}, we can quickly reach to the stage where
    \begin{align}
        |\sigma_n^*(x)-\sigma^*(x)| = O(|F_{\mu_n}(x,\sigma_n^*(x))-F_{\mu}(x,\sigma_n^*(x))|) \to 0.
    \end{align}
    Moreover, we know that $\mu$ has compact support and $x \mapsto \sigma^*(x)$ is smooth, $\sigma^*(X_i)$ is bounded above and below, meaning that  $\sigma_n^*(X_i)$ is bounded uniformly from above and below as well. This completes the proof in the original paper.
\end{proof}
