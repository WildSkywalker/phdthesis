\section{Setup}
Here goes the real math. Recall that this document is dedicated to the
reconstruction of \cite{auffinger-tsne} with low-dimension kernel $g$ replaced
by a more general kernel. This would serve as a great way to understand the
material and help with the completion of the original project. For the sake of
simplicity, we will use the exact same notation as in \cite{auffinger-tsne} and
skip all the definitions. Recall in \cite{auffinger-tsne}, $g$ is defined as
\begin{align}
    g(y,y') = \frac{1}{1+\norm{y-y'}^2}
\end{align}
With all these being said, we will first start by giving our general low dimension kernel $g$ some general assumptions, and modify it as we move on. We first require that $g(y,y') = k (\norm{y-y'})$ such that:

1. $k: \mathbb{R}^+ \to \mathbb{R}^+$ is decreasing and bounded.

2. $k$ satisfies
\begin{align}
    \iint_{(0,\infty)^2} k(x,x') dxdx' < \infty.
\end{align}
We will kick off this warmup by listing all the important lemmas and results in order, and prove our own versions when the need arises. The final goal is to prove a same set of results as in \cite{auffinger-tsne} with the $y$ kernel replaced.
Recall that
\begin{align}
    F_{\rho, \mu}(x,\sigma) = & - \int \frac{\norm{x-x'}^2}{2\sigma^2} \frac{\exp(-\norm{x-x'}^2/2\sigma^2)}{\int \exp(-\norm{x-x'}^2/2\sigma^2)d\mu(x')}d\mu(x') \nonumber \\
                              & - \log \left(\int \exp(-\norm{x-x'}^2/2\sigma^2)\right) +\log \rho.
\end{align}
\begin{proposition}
    If $\mu$ has a $C^1$ density $f$ such that both $f$ and $\norm{\nabla f}$ are bounded, then the equation $F_{\rho, \mu}(x,\sigma)=0$ has a unique solution $\sigma^*_{\rho,\mu}(x)$ for all $\rho \in (0,1)$ and $x \in \mathbb{R}^d$. Moreover, $x \mapsto \sigma^*_{\rho,\mu}(x)$ is smooth.
\end{proposition}
\begin{proof}
    This proposition is about the kernel of $x$. The proof stays unchanged from \cite{auffinger-tsne}.
\end{proof}
The next couple of lemmas are general results in section 3 of \cite{auffinger-tsne}. We will omit the proof because they are not directly discussing the quantities in the t-sne algorithm.  Recall that $X$ or the corresponding probability measure $\mu$ is called sub-Gaussian if $\p(X>t) \leq 2 \exp(-Ct^2)$ for some positive constant $C$.
\begin{lemma} \label{Gaussian_lemma}
    Let $\nu$ be sub-Gaussian with second moment equal to 1 and suppose $X \in \mathbb{R}^d$ has i.i.d. entries drawn from $\nu$. Then there exists constant $C, c_1$ such that
    \begin{align}
        \p(\norm{X} \geq \sqrt{d} + t) \leq Ce^{-c_1t^2}.
    \end{align}
\end{lemma}
\begin{lemma} \label{max_iid_lemma}
    Let $\mu$ be sub-Gaussian and $c_1$ be the constant from lemma \ref{Gaussian_lemma}. Let $X_1, \dots, X_n \in \mathbb{R}^d$ have i.i.d. entries drawn from $\mu$. Then there exists a constant $C$ such that
    \begin{align}
        \p \left (\max_{i=1,\dots,n}\norm{X_i} \geq \sqrt{\frac{\log n}{c_1}}+\sqrt{d}+t \right ) \leq Ce^{-c_1t^2}.
    \end{align}
\end{lemma}
\begin{lemma} \label{approx_lemma}
    Let $\delta>0$ be constant, $K_n = [\delta,\sqrt{n}]$ and $h: \mathbb{R}^d \times \mathbb{R}^d \times \mathbb{R} \to \mathbb{R}$. Suppose the following hold: \\
    1. $\mu$ is sub-Gaussian; \\
    2. $h(x,x',\sigma)$ and $\nabla_x h(x,x',\sigma)$ are bounded for al $x,x'$ and all $\delta \in K_n$;\\
    3. $h(x,x',\sigma) \to 0$ as $\norm{x-x'}/\sigma \to \infty$.\\
    Let $X_1,\dots,X_n$ be i.i.d. random variables drawn from $\mu$. Let $\mu_n = (1/n)\sum_i \delta_{X_i}$. Then for any $\epsilon > 0$ with probability at least $1 - e^{-c\epsilon^2n}$ we have
    \begin{align}
        \sup_{\sigma \in K_n,x} \left|\int h(x,x',\sigma)d\mu_n(x) - \int h(x,x',\sigma)d\mu(x')\right| \leq \epsilon + O\left(\sqrt{\frac{\log n}{n}}\right).
    \end{align}
\end{lemma}
Define
\begin{align}
    f_{\sigma}(x,x') = \exp(-\norm{x-x}^2/2\sigma^2), \quad g_{\sigma} = \frac{\norm{x-x}^2}{2\sigma^2} \exp(-\norm{x-x'}^2/2\sigma^2).
\end{align}
The next two lemmas use similar techniques to prove.
\begin{lemma}
    Let $c_1$ be the decay constant in lemma \ref{Gaussian_lemma} and let $\delta >0$. Then almost surely in $X_1,\dots,X_n$,
    \begin{align}
        \sup_{\substack{\sigma \in [\delta, \sqrt{n}] \\ i = 1,\dots,n}}\left|\frac{\int g_{\sigma}(X_i,x')d\mu_n(x')}{\int f_{\sigma}(X_i,x')d\mu_n (x')} - \frac{\int g_{\sigma}(X_i,x')d\mu(x')}{\int f_{\sigma}(X_i,x')d\mu (x')}\right| = O(n^{8/c_1 \delta^2 -1/2}\sqrt{\log n}).
    \end{align}
\end{lemma}
\begin{lemma}\label{pair_lemma_2}
    Let $c_1$ be the decay constant in lemma $\ref{Gaussian_lemma}$ and let $\delta>0$ be such that $c_1\delta^2>8$. Then almost surely in $X_1,\dots,X_n$,
    \begin{align}
        \sup_{\substack{\sigma \in [\delta, \sqrt{n}] \\ i = 1,\dots,n}} \left|\log \left(\int f_{\sigma}(X_i,x')d\mu_n (x')\right) - \log \left(\int f_{\sigma}(X_i,x')d\mu (x')\right)\right| = O(n^{4/c_1\delta^2 -1/2}\sqrt{\log n})
    \end{align}
\end{lemma}
For this pair of lemmas, \cite{auffinger-tsne} provides the proof to the first lemma. We will provide the second for the sake of exercise.
\begin{proof}[Proof of lemma \ref{pair_lemma_2}]
    First, notice that for arbitrary $x>0$, the following elementary inequality holds:
    \begin{align}
        \log(x) \leq x-1.
    \end{align}
    Therefore, the lemma can be proved if we are able to show that
    \begin{align}
        \left|\frac{\int f_{\sigma}(X_i,x')d\mu_n (x')}{\int f_{\sigma}(X_i,x')d\mu (x')} - 1 \right| = & \left|\frac{\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')}{\int f_{\sigma}(X_i,x')d\mu (x')} \right|\nonumber \\
        =                                                                                               & O(n^{4/c_1\delta^2 -1/2}\sqrt{\log n}).
    \end{align}
    We can achieve that by bounding the numerator and the denominator of the previous formula. On one hand, consider set
    \begin{align}
        E_1' = \left \{ \max_{i=1,\dots,n}\norm{X_i} \geq \sqrt{\alpha+\frac{1}{c_1}} \sqrt{\log n}+\sqrt{d}\right \}
    \end{align}
    Setting $t = \sqrt{\alpha \log n}$ in lemma \ref{max_iid_lemma}, we have $\p(E_1') \leq Cn^{-c_1\alpha}$. To absorb the $\sqrt{d}$ term, we can find $\epsilon_n \to 0$ such that
    \begin{align}
        E_1 = \left \{ \max_{i=1,\dots,n}\norm{X_i} \geq \sqrt{\alpha+\frac{1}{c_1}+\epsilon_n} \sqrt{\log n}\right \}
    \end{align}
    with $\p(E_1) \leq Cn^{-c_1\alpha}$.
    Because the maximum of norm of each $X_i$ is restricted, on $E_1^c$ we have for all $i,j$, that
    \begin{align}
        \norm{X_i-X_j}^2 \leq 4 \left(\alpha + \frac{1}{c_1} +\epsilon_n\right) \log n =:4\gamma \log n.
    \end{align}
    Then for $\sigma \in [\delta, \sqrt{n}]$, on $E_1^c$ for all $i$ we have
    \begin{align} \label{log_esti_1}
        \int f_{\sigma}(X_i,x')d\mu(x') \geq (1-Cn^{-c_1\alpha})\exp \left(-\frac{2\gamma \log n}{\delta^2}\right) \geq \frac{1}{2} \exp \left(-\frac{2\gamma \log n}{\delta^2}\right) =\frac{1}{2}n^{-2\gamma/\delta^2}.
    \end{align}
    This helps us bound the denominator. On the other hand, by lemma \ref{approx_lemma}, we are able to find set $E_2$ with $\p(E_2) \leq e^{-c\epsilon^2 n}$, such that, on $E^c_2$,
    \begin{align} \label{log_esti_2}
        \sup_{\substack{\sigma \in [\delta,\sqrt{n}] \\ i = 1,\dots,n}} \left|\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')\right| \leq \epsilon + O\left(\frac{\sqrt{\log n}}{\sqrt{n}}\right).
    \end{align}
    Therefore, we can combine (\ref{log_esti_1}) and (\ref{log_esti_2}) and conclude that on $E_1^c \cap E_2^c$,
    \begin{align}
        \left|\frac{\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')}{\int f_{\sigma}(X_i,x')d\mu (x')} \right| \leq C\epsilon n^{2\gamma/\delta^2}+O(\sqrt{\log n} \cdot n^{-1/2 + 2\gamma/\delta^2}).
    \end{align}
    In order for the RHS to converge to zero, we need
    \begin{align}
        \epsilon n^{2\gamma/\delta^2} \to 0, \quad 1/2 - 2\gamma/\delta^2 > 0.
    \end{align}
    Moreover, we have $\p(E_1 \cup E_2) \leq Cn^{-c_1\alpha} + e^{-c \epsilon^2 n}$. Set $\epsilon = n^{-\eta}$ for some $\eta >0$. In order to get almost sure convergence using Borel-Cantelli, we also need
    \begin{align}
        c_1 \alpha>1, \quad 1-2\eta>0.
    \end{align}
    Hence for the entire lemma to work we need the following four inequalities to hold:
    \begin{align}
        \frac{2\gamma}{\delta^2}-\eta<0, \quad \frac{2\gamma}{\delta^2}-1/2<0, \quad c_1 \alpha>1, \quad 1-2\eta>0.
    \end{align}
    The last inequality implies that $\eta<1/2$ so the first inequality implies the second. Now, recall that
    \begin{align}
        \gamma := \alpha + \frac{1}{c_1} +\epsilon_n.
    \end{align}
    The collection of inequalities is equivalent to
    \begin{align} \label{inequ_collection}
        2\left(\alpha + \frac{1}{c_1} +\epsilon_n\right)<\eta \delta^2, \quad c_1 \alpha>1, 0<\eta<1/2.
    \end{align}
    Here, we are free to choose $\alpha$ and $\eta$, but $c_1$ and $\delta$ are fixed. Now let
    \begin{align}
        \alpha = \frac{1}{c_1}+\epsilon, \quad \eta = \frac{1}{2} - \epsilon.
    \end{align}
    Then the second and third inequalities in (\ref{inequ_collection}) are satisfied. The first is equivalent to
    \begin{align}
        \frac{4}{c_1 \delta^2}+\epsilon<\frac{1}{2}-\epsilon
    \end{align}
    as $\epsilon_n \to 0$ as $n \to \infty$. In addition, since $\epsilon$ is arbitrary, we have again, equivalently,
    \begin{align}
        \frac{4}{c_1 \delta^2} - \frac{1}{2}<0.
    \end{align}
    When this is satisfied, we have:
    \begin{align}
        \sup_{\substack{\sigma \in [\delta,\sqrt{n}] \\ i = 1,\dots,n}} \left|\int f_{\sigma}(X_i,x')d\mu_n (x')-\int f_{\sigma}(X_i,x')d\mu (x')\right| =O(n^{4/c_1\delta^2 -1/2}\sqrt{\log n}).
    \end{align}
    The proof is thus finished.
\end{proof}
Then combining the previous result we have an important convergence:
\begin{proposition}
    Let $\mu$ be sub-Gaussian and $c_1$ be the constant from lemma \ref{Gaussian_lemma}. $X_1,\dots, X_n$ are i.i.d. random variables with probability measure $\mu$ and $\mu_n = \frac{1}{n} \sum \delta_{X_i}$ is the empirical measure. Additionally, set $\delta>0$ to be a constant such that $c_1 \delta^2>16$. Then, there exists $\epsilon>0$, such that almost surely in $X_1,\dots,X_n$:
    \begin{align}
        \sup_{\substack{\sigma \in [\delta,\sqrt{n}] \\i=1,\dots,n}}|F_{\mu_n}(X_i,\sigma)-F_{\mu}(X_i,\sigma)| = O(n^{-\epsilon}).
    \end{align}
\end{proposition}
Let us summarize the important results upon this point, which will be used later:

There exists $\gamma = 2/c_1 +\epsilon$ such that the following holds almost
surely

(1)$\max_{i=1,\dots,n}\norm{X_i} \leq \sqrt{\gamma \log n}$;

(2)$\max_{i,j}\norm{X_i-X_j}^2 \leq 4 \gamma \log n$;

(3)For $\sigma\in [\delta,\sqrt{n}]$,
\begin{align}
    \int \exp(-\norm{X_i-x'}^2/2\sigma^2)d\mu_n(x),\int \exp(-\norm{X_i-x'}^2/2\sigma^2)d\mu(x') \geq Cn^{-2\gamma/\sigma^2}.
\end{align}
Next couple of lemmas in \cite{auffinger-tsne} are about lower about on $\sigma^*_{\rho,\mu}$. Recall that $\sigma^*_{\rho,\mu}$ is the unique solution of the equation $F_{\rho,\mu}(x,\sigma^*_{\rho,\mu})=0$. For simplicity of notation, we write $\sigma^*(x) = \sigma^*_{\rho,\mu}(x)$ and $\sigma^*_n(x) = \sigma^*_{\rho,\mu_n}(x)$.
\begin{lemma} \label{sigma_lower_bdd_lemma}
    If $\mu$ satisfies the condition in lemma 1 of \cite{auffinger-tsne}, then $\sigma^*(x)$ and $\sigma^*_n(x)$ are uniformly bounded from below in both $i$ and $n$ almost surely.
\end{lemma}
\begin{proposition} \label{sigma_diff_prop}
    Suppose $\mu$ has compact support and $C^1$ density. Then almost surely, as $n \to \infty$,
    \begin{align}
        \sup_{i=1,\dots,n}|\sigma_n^*(X_i)-\sigma^*(X_i)| \to 0.
    \end{align}
\end{proposition}
\begin{proof}
    By \cite{auffinger-tsne}, we can quickly reach to the stage where
    \begin{align}
        |\sigma_n^*(x)-\sigma^*(x)| = O(|F_{\mu_n}(x,\sigma_n^*(x))-F_{\mu}(x,\sigma_n^*(x))|) \to 0.
    \end{align}
    Moreover, we know that $\mu$ has compact support and $x \mapsto \sigma^*(x)$ is smooth, $\sigma^*(X_i)$ is bounded above and below, meaning that  $\sigma_n^*(X_i)$ is bounded uniformly from above and below as well. This completes the proof in the original paper.
\end{proof}

\section{More Lemmas}
Starting this section we are gradually approaching our point of work. We need
to list a few more lemmas before that. To make notations cleaner,
\cite{auffinger-tsne} introduced some abbreviation in section 4.2. We will use
all of them by default.
\begin{lemma}
    Any sequence in $\tilde{\mathcal{P}}_X$ is tight.
\end{lemma}
Recall Prokhorov's theorem:
\begin{theorem*}[Prokhorov's Theorem]
    If $\mu_n$ is a tight sequence of measures in Euclidean space. There exists a subsequence $\mu_{n_k}$ and a measure $\mu$ such that $\mu_{n_k}$ converges to $\mu$ weakly.
\end{theorem*}
We immediately get the following lemma:
\begin{lemma} \label{mu_star_lemma}
    There exits $\mu^*$ such that
    \begin{align}
        I_{\rho}(\mu^*) = \inf_{\mu \in \tilde{\mathcal{P}}_X} I_{\rho}(\mu)
    \end{align}
\end{lemma}
Recall that $I_{\rho}(\mu)$ is the continuous version of $L_{n,\rho}(X,Y)$. It only depends on the measure $\mu$ and the perplexity constant $\rho$. We have not reached the part that requires we have a common perplexity, but from now on we should pay close attention as it is one of the core setup of the original paper.
\begin{lemma}
    As $n \to \infty$, for some $\epsilon>0$, we have
    \begin{align}
        |f_n(X_i,X_j)-f(X_i,X_j)| = O(n^{-\epsilon}).
    \end{align}
\end{lemma}
\begin{proof}
    There are typos in the original proof so we prove this lemmas again despite it being very easy and elementary. We compute directly
    \begin{align}
             & |f_n(x,x')-f(x,x')| = |\exp(-\norm{x-x}^2/2\sigma_n(x)^2)-\exp(-\norm{x-x}^2/2\sigma(x)^2)| \nonumber                                                          \\
        =    & \left| \exp \left(\frac{|\sigma_n(x)^2-\sigma(x)^2|\norm{x-x}^2}{2\sigma_n(x)^2 \sigma(x)^2}\right)-1\right| \cdot \exp(-\norm{x-x'}^2/2\sigma(x)^2) \nonumber \\
        \leq & \left| \exp \left(\frac{|\sigma_n(x)^2-\sigma(x)^2|\norm{x-x}^2}{2\sigma_n(x)^2 \sigma(x)^2}\right)-1\right|.
    \end{align}
    Recall that the sum of two sub-Gaussian random variables is still sub-Gaussian. If $\mu_X$ is sub-Gaussian, let $C$ be the constant of decay for $X_i-X_j$ and we have
    \begin{align} \label{distance_esti}
        \p(\norm{X_i-X_j}^2 \geq \log n) \leq 2 \exp(-\log n /C^2) = 2 \exp\left(\frac{1}{c^2}\right) \cdot \frac{1}{n} \to 0.
    \end{align}
    This means $\norm{X_i-X_j}^2 = O(\log n)$ almost surely. On the other hand, if $\mu_X$ has compact support then $\norm{X_i-X_j}$ will be bounded, even easier. Moreover, since $\sigma_n$ and $\sigma$ are both uniformly bounded below by lemma (\ref{sigma_lower_bdd_lemma}), by setting $x=X_i$ and $x'=X_j$ we get
    \begin{align}
             & \left| \exp \left(\frac{|\sigma_n(x)^2-\sigma(x)^2|\norm{x-x}^2}{2\sigma_n(x)^2 \sigma(x)^2}\right)\right| \nonumber                  \\
        \leq & \frac{1}{\sigma_n(x) \sigma(x)}\left(\frac{1}{\sigma_n(x)} + \frac{1}{\sigma(x)}\right)|\sigma_n(x)-\sigma(x)|\norm{x-x'}^2 \nonumber \\
        =    & O(|\sigma_n(x)-\sigma(x)|\norm{x-x'}^2) \nonumber                                                                                     \\
        =    & O(|\sigma_n(x)-\sigma(x)|)O(\log n) \nonumber                                                                                         \\
        =    & O(n^{-\epsilon})O(\log n) \nonumber                                                                                                   \\
        =    & O(n^{-\epsilon}),
    \end{align}
    by proposition (\ref{sigma_diff_prop}). This finishes the proof.
\end{proof}
One thing we should highlight here is that we are abusing the notation of $\sigma$ and $\sigma_n$. They are defined as the unique solution to $F=0$. However, the definition of $\sigma_i$ is not abused and we will later see why such abuse makes sense.
\begin{lemma} \label{integrand_lemma_2}
    If $c_1 \delta >32$, then there exists $\epsilon>0$ such that as $n \to \infty$, we have
    \begin{align}
        \sup_{i,j}|p'_n(X_i,X_j)-p'(X_i,X_j)| & = O(n^{-\epsilon}), \nonumber \\
        \sup_{i,j}|p_n(X_i,X_j)-p(X_i,X_j)|   & = O(n^{-\epsilon}).
    \end{align}
\end{lemma}
\begin{proof}[Supplementary Proof]
    The overall methodology of the original proof in \cite{auffinger-tsne} is natural and clear. However, it can still be slightly confusing as it is missing the actual values we assign to $\epsilon$. In the original paper, we can take, for some constant $k>0$, that
    \begin{align}
        n^{-\epsilon} = n^{8/c_1\delta^2 -1/2} (\log n)^k.
    \end{align}
    This comes from the detailed computations in the original proof and the quantity above is the slowest decaying speed, the 'worst case scenario' given all the conditions. Plug this into the computations and everything is clear. The proof uses a little bit extra than just the bounds in section 3.1.

\begin{proof}
    Fortunately, the first proof is almost identical to the original one. Let $k \in \{1,\cdots, n\}$. We are going to show that
    \begin{align}
        \lim_{|y_k| \to \infty} L_{n,\rho}(X,y) = \infty.
    \end{align}

    \begin{align}
        L_{n,\rho}(X,y) = \sum_{i \neq j} p_{ij}\log \left(\frac{p_{ij}}{q_{ij}}\right) = \sum_{i \neq j} p_{ij}\log p_{ij} - \sum_{i \neq j} p_{ij}\log q_{ij}.
    \end{align}

    \begin{align}
        \sum_{i \neq j} p_{ij}\log (q_{ij}) = \sum_{i \neq j} p_{ij}\log \left(\frac{g(y_i,y_j)}{\sum_{m \neq l}g(y_m,y_l)}\right) \to -\infty.
    \end{align}

    \begin{align}
             & -\sum_{i \neq j} p_{ij} log \left(\sum_{m \neq l}g(y_m,y_l) \right) = -\log \left(\sum_{m \neq l}g(y_m,y_l) \right) \nonumber \\

    \end{align}
    On the other hand, we have $g(y_i,y_k) = k (\norm{y_i-y_k}) \to 0$ as $|y_k| \to \infty$ because $\norm{y_i-y_k} \to \infty$ and $k$ satisfies

    \end{align}
    Hence, the minimum of the $L_{n,\rho}(X,y)$ is achieved on a bounded set, the lemma follows by the continuity of $L_{n,\rho}(X,y)$.

\subsection*{Partial Derivative of the Relative Entropy}
Before we move on, we would like to take a small deviation and compute the
partial derivative of the relative entropy. This is important because it will
impact the property of the set $\tilde{\mathcal{P}}_X$.
\begin{lemma} \label{derivative_lemma}
    Let $X \in (\mathbb{R}^d)^n$ and $Y = (y_1,\dots,y_n)\in (\mathbb{R}^s)^n$ be the input and output of the t-SNE algorithm, then we have
    \begin{align}
        \frac{\partial L_{n, \rho}(X,Y)}{\partial y_i} = -\sum_{j=1, j \neq i}^n (p_{ij}-q_{ij}) \frac{\frac{\partial}{\partial y_i}g(y_i,y_j)}{g(y_i,y_j)}.
    \end{align}
\end{lemma}
\begin{proof}
    By definition,
    \begin{align}
          & L_{n, \rho}(X,Y) = \sum_{i \neq j} p_{ij} \log \left(\frac{p_{ij}}{q_ij}\right) \nonumber \\
        = & \sum_{i \neq j} p_{ij} \log p_{ij} - \sum_{i \neq j} p_{ij} \log q_{ij}.
    \end{align}
    The first summation does not depend on $y_i$ and will vanish when taking the partial derivative. Moreover, we can further breakdown the second summation:
    \begin{align}
          & - \sum_{i \neq j} p_{ij} \log q_{ij} = - \sum_{i \neq j} p_{ij} \log g(y_i,y_j) + \sum_{i \neq j} p_{ij} \log \left(\sum_{k \neq l} g(y_k,y_l) \right)\nonumber \\
        = & - \sum_{i \neq j} p_{ij} \log g(y_i,y_j) + \log \left(\sum_{k \neq l} g(y_k,y_l)\right).
    \end{align}
    Taking derivative of the last expression and note that only terms that contain $y_i$ remains. This means
    \begin{align}
        \frac{\partial L_{n, \rho}(X,Y)}{\partial y_i} = & \sum_{j \neq i, j = 1}^n \left(-p_{ij}\frac{\frac{\partial}{\partial y_i}g(y_i,y_j)}{g(y_i,y_j)} + \frac{1}{\sum_{k \neq l} g(y_k,y_l)} \frac{\partial}{\partial y_i} g(y_i,y_j)\right) \nonumber                            \\
        =                                                & \sum_{j \neq i, j = 1}^n \left(-p_{ij}\frac{\frac{\partial}{\partial y_i}g(y_i,y_j)}{g(y_i,y_j)} + \frac{g(y_i,y_j)}{\sum_{k \neq l} g(y_k,y_l)} \frac{\frac{\partial}{\partial y_i}g(y_i,y_j)}{g(y_i,y_j)}\right) \nonumber \\
        =                                                & -\sum_{j=1, j \neq i}^n (p_{ij}-q_{ij}) \frac{\frac{\partial}{\partial y_i}g(y_i,y_j)}{g(y_i,y_j)}.
    \end{align}
\end{proof}
We can easily do a sanity test by letting $g(y_i,y_j) = (1+\norm{y_i-y_j}^2)^{-1}$ and get formula (A.1) in \cite{auffinger-tsne}.

        \sum_{i \neq j} p_{ij} \log \left(\frac{p_{ij}}{q_{ij}}\right) = \sum_{i \neq j} p_{ij} \log p_{ij} - \sum_{i \neq j} p_{ij} \log q_{ij}.

        \liminf_{n \to \infty} p_n(x,x')\log g(y,y') d\mu_n d\mu_n \geq  \iint p(x,x')\log g(y,y') d\mu d\mu.
    \end{align}

        \liminf_{n \to \infty} \iint g(y,y') d\mu_n d\mu_n - \frac{1}{n} \geq \iint g(y,y')d\mu d\mu.
    \end{align}

Moreover, because of our assumption on the kernel $g$, the integrals on the RHS of the last two inequalities converge. In addition, since $\log$ is a continuous and increasing function, (\ref{fatou_esti_2}) implies

    \end{align}
    The proof is finished by combining (\ref{entropy_esti_2}), (\ref{fatou_esti_1}), (\ref{fatou_esti_3}) and lemma \ref{entropy_lemma_1}. The $2 \log n$ cancelled out because of (\ref{entropy_esti}).
\end{proof}

        \limsup_{n \to \infty} \inf_{Y \in (\mathbb{R}^s)^n} L_{n,\rho}(X,Y) \leq I_{\rho}(\mu^*).
    \end{align}
\end{proposition}
